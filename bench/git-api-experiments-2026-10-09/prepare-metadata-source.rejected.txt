  "src/shell/commands/git-metadata-reads.ts",
  `import { VfsError } from "../../core/errors.js";
import { FsStats } from "../../fs/stats.js";
import type { ShellCommandContext } from "../types.js";
interface MetadataRequest { path: string; resolve: (stat: FsStats) => void; reject: (error: unknown) => void; }
/** Coalesce only overlapping requests; never retain a completed snapshot. */
export class GitMetadataReads {
  private readonly pending: MetadataRequest[] = [];
  constructor(private readonly context: ShellCommandContext) {}
  lstat(path: string): Promise<FsStats> {
    return new Promise((resolve, reject) => {
      this.pending.push({path, resolve, reject});
      if (this.pending.length === 1) queueMicrotask(() => this.flush());
    });
  }
  private flush(): void {
    const pending = this.pending.splice(0);
    for (let offset = 0; offset < pending.length; offset += 128) this.resolveBatch(pending.slice(offset, offset + 128));
  }
  private resolveBatch(batch: MetadataRequest[]): void {
    try {
      if (this.context.signal.aborted) throw new VfsError("ECANCELED", "Git execution was cancelled");
      this.context.budget.checkDeadline();
    } catch(error) { for (const request of batch) request.reject(error); return; }
    const fs = this.context.fileSystem;
    try {
      if (fs.statMany === undefined) throw new VfsError("ENOTSUP", "metadata batching unavailable");
      const stats = fs.statMany(batch.map(request => request.path), {follow:false});
      for (const [i,request] of batch.entries()) { const stat = stats[i]; if(stat === undefined) throw new VfsError("EIO", "incomplete metadata batch"); request.resolve(new FsStats(stat)); }
    } catch {
      for (const request of batch) {
        try { request.resolve(new FsStats(fs.lstat(request.path))); } catch (error) { request.reject(error); }
      }
    }
  }
}
`,
