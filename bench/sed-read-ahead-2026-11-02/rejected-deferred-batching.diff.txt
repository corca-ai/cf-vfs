+const BATCH_ENTRIES = 64;
+const BATCH_BYTES = 1024 * 1024;

+/** One whole-file rewrite, guarded by the token its source was read at. */
+export type InPlaceWrite = WriteFilesEntry & { readonly body: string };
+
+interface PendingWrite {
+  readonly entry: InPlaceWrite;
+  /** The operand as the caller named it, for its diagnostic. */
+  readonly name: string;
+}
+
+/**
+ * Guarded whole-file rewrites a command makes over many operands, published
+ * together.
+ *
+ * Each rewrite would otherwise be its own transaction and its own statements.
+ * They collect here and go out as one `writeFiles` batch, which either
+ * applies every rewrite or none. A batch that fails is written again one file
+ * at a time, so each operand meets exactly the outcome and the diagnostic it
+ * would have met alone, in operand order. Callers flush before reporting
+ * anything else and before reading a path that is still pending, so neither
+ * output order nor what a later operand reads can tell the difference.
+ */
+export class InPlaceWrites {
+  readonly #context: ShellCommandContext;
+  readonly #report: (name: string, error: unknown) => Promise<void>;
+  #pending: PendingWrite[] = [];
+  #paths = new Set<string>();
+  #bytes = 0;
+
+  constructor(
+    context: ShellCommandContext,
+    report: (name: string, error: unknown) => Promise<void>,
+  ) {
+    this.#context = context;
+    this.#report = report;
+  }
+
+  /** Whether a rewrite of `path` is waiting to be published. */
+  pending(path: string): boolean {
+    return this.#paths.has(path);
+  }
+
+  /** Queues a rewrite; resolves to whether a flush it caused reported a failure. */
+  async add(entry: InPlaceWrite, name: string): Promise<boolean> {
+    this.#pending.push({ entry, name });
+    this.#paths.add(entry.path);
+    this.#bytes += entry.body.length;
+    if (this.#pending.length < BATCH_ENTRIES && this.#bytes < BATCH_BYTES) return false;
+    return await this.flush();
+  }
+
+  /** Publishes everything queued; resolves to whether any rewrite failed. */
+  async flush(): Promise<boolean> {
+    const batch = this.#pending;
+    this.#pending = [];
+    this.#paths = new Set();
+    this.#bytes = 0;
+    if (batch.length === 0 || (await this.#publishTogether(batch))) return false;
+    return await this.#publishEach(batch);
+  }
+
+  /** One atomic batch; false when it was not tried or changed nothing. */
+  async #publishTogether(batch: readonly PendingWrite[]): Promise<boolean> {
+    const { budget, fileSystem } = this.#context;
+    // A batch that would cross the mutation limit is left to single writes,
+    // which stop at the exact file the limit stops at.
+    const fits = (budget.remainingMutations?.() ?? 0) >= batch.length;
+    if (batch.length < 2 || fileSystem.writeFiles === undefined || !fits) return false;
+    try {
+      await fileSystem.writeFiles(batch.map((write) => write.entry));
+      return true;
+    } catch (error) {
+      if (error instanceof VfsError) return false;
+      throw error;
+    }
+  }
+
+  async #publishEach(batch: readonly PendingWrite[]): Promise<boolean> {
+    let failed = false;
+    for (const { entry, name } of batch) {
+      try {
+        await this.#context.fileSystem.writeFile(entry.path, entry.body, {
+          ...(entry.ifMutationToken === undefined
+            ? {}
+            : { ifMutationToken: entry.ifMutationToken }),
+          disposition: "replace",
+          ...(entry.mode === undefined ? {} : { mode: entry.mode }),
+        });
+      } catch (error) {
+        await this.#report(name, error);
+        failed = true;
+      }
+    }
+    return failed;
+  }

+function refuseBatches(inner: object, property: string | symbol): unknown {
+  if (property === "readMany") return undefined;
+  if (property === "writeFiles") {
+    return () => Promise.reject(new VfsError("ENOTSUP", "batch writes are unavailable"));
+  }
+  const value: unknown = Reflect.get(inner, property);
+  return typeof value === "function" ? value.bind(inner) : value;
+}

+/**
+ * A filesystem whose views offer no bulk reads and refuse batch writes, so a
+ * batch falls back to writing each file alone.
+ */
+function oneAtATime(fs: NodeSqlFileSystem): NodeSqlFileSystem {
+  return new Proxy(fs, {
+    get(inner, property) {
+      if (property !== "forCredentials") return refuseBatches(inner, property);
+      return (...args: Parameters<NodeSqlFileSystem["forCredentials"]>) =>
+        new Proxy(inner.forCredentials(...args), { get: refuseBatches });
+    },
+  });
+}
