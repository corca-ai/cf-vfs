import { runInDurableObject } from "cloudflare:test";
import { env } from "cloudflare:workers";
import { expect, it } from "vitest";
import { DurableObjectFileSystem } from "../src/vfs/do-sql.js";
import { meterSqlStorage } from "./metered-sql.js";
import { DurableObjectFileSystem as BaselineFileSystem } from "./three-hour-2026-10-10/compiled/baseline/src/vfs/do-sql.js";

it("compares actual SQLite costs of fused credential-bound reads", async () => {
  const rows = await runInDurableObject(env.VFS_TEST.getByName("fused-read-cost"), async (_, state) => {
    const rows = [];
    for (const [version, Constructor] of [["baseline", BaselineFileSystem], ["candidate", DurableObjectFileSystem]] as const) {
      await state.storage.deleteAll();
      const meter = meterSqlStorage(state.storage);
      const fs = new Constructor(meter.storage);
      for (const depth of [1, 16, 64]) {
        const path = `/${Array(depth).fill("d").join("/")}/file`;
        await fs.writeFile(path, "x".repeat(768), { createParents: true });
        const view = fs.forCredentials({ uid: 1000, gid: 1000 });
        for (const operation of ["whole", "eof"] as const) {
          meter.reset();
          const read = view.readFile(path, operation === "eof" ? { range: { offset: Number.MAX_SAFE_INTEGER } } : undefined);
          await read.stream.cancel();
          rows.push({ version, depth, operation, statements: meter.statements, rowsRead: meter.rowsRead, rowsWritten: meter.rowsWritten });
        }
      }
    }
    return rows;
  });
  console.info(`FUSED READ COST ${JSON.stringify(rows)}`);
  for (const before of rows.filter((row) => row.version === "baseline")) {
    const after = rows.find((row) => row.version === "candidate" && row.depth === before.depth && row.operation === before.operation);
    expect(after).toBeDefined();
    expect(after?.statements).toBe(before.statements - 1);
    expect(after?.rowsRead).toBeLessThanOrEqual(before.rowsRead);
    expect(after?.rowsWritten).toBe(0);
  }
});
