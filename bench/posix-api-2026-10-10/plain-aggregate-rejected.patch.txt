+           MIN(CASE WHEN kind <> 'directory' OR ${allowed} = 0 THEN path END) AS invalid,
+           MIN(CASE WHEN kind <> 'directory' THEN path END) AS nondirectory
+         FROM vfs_entries parent INDEXED BY vfs_entries_path
+         WHERE path IN (SELECT value FROM json_each(?))
